# Нативные библиотеки, которых НЕТ в облегчённой сборке Android (Э7).
#
# 🔴 ОДИН СПИСОК НА ДВА МЕСТА. По нему и Gradle выбрасывает библиотеки из
# пакета (android/app/build.gradle.kts), и скрипт выпуска проверяет готовый
# APK. Когда списки были двумя, в сборке 12.10.2026 остались libwhisper.so и
# libparakeet.so: в исключениях их не было, а проверка искала только те имена,
# что уже были в исключениях, — то есть подтверждала сама себя.
#
# Строки — имена файлов, допускается `*`. Пустые строки и строки с `#`
# пропускаются.

# Перевод сообщений на устройстве и определение языка (ML Kit)
libtranslate_jni.so
liblanguage_id_l2c_jni.so

# Расшифровка голосовых на устройстве (whisper.cpp и его движок)
libwhisper*.so
libggml*.so
libparakeet.so

# Обработка видео (FFmpeg)
#
# 🔴 СО ЗВЁЗДОЧКОЙ, И ЭТО НЕ ПЕРЕСТРАХОВКА. На armeabi-v7a FFmpegKit везёт
# те же библиотеки под ДРУГИМИ именами — `libavcodec_neon.so`,
# `libffmpegkit_armv7a_neon.so` и так далее. Со списком точных имён первая
# 32-разрядная облегчённая сборка (12.10.2026) увезла 23,5 МиБ FFmpeg,
# формально пройдя проверку: проверка сверялась с этим же списком.
libavcodec*.so
libavdevice*.so
libavfilter*.so
libavformat*.so
libavutil*.so
libswresample*.so
libswscale*.so
libffmpegkit*.so
